# Bam v0.2 Part A - Part 2: Detail, Watch and Overlays Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fill the main panel with build detail and a navigable stage and job tree, drive it live from `app.Watch`, add the Presets panel's drill-down, and add the server, project and branch overlays.

**Architecture:** The watch is the only polling path in the UI and at most one runs: `Model` holds its `context.CancelFunc` and cancels it whenever the build selection changes. `app.Event` values arrive as `watchEventMsg` through the standard bubbletea one-message-at-a-time channel pattern. Overlays are a single `overlay` field plus one `listState` reused for whichever picker is open.

**Tech Stack:** Go 1.26+, bubbletea v1.3.10, lipgloss v1.1.0, bubbles v0.21.1, testify.

**Spec:** `docs/superpowers/specs/2026-09-18-bam-v0.2-tui-navigate-design.md`, especially §4.1, §5 and §3.2. Read part 1's plan first; this part assumes its tasks are done and green.

```go
package tui

import (
	"fmt"
	"strings"
	"time"

	"github.com/r0jjames/bam-cli/internal/provider"
	"github.com/r0jjames/bam-cli/internal/view"
)

type treeKind int

const (
	rowStage treeKind = iota
	rowJob
)

// treeRow is one visible line of the stage and job tree. Stage and Job are
// indexes into the build, so a row survives a refresh that reorders nothing.
type treeRow struct {
	Kind     treeKind
	Stage    int
	Job      int
	Name     string
	State    provider.State
	Duration time.Duration
	Key      string // job result key; empty for a stage
	URL      string
}

// buildTree flattens the build into the rows to draw, honouring which stages
// are expanded.
func buildTree(b provider.Build, expanded map[string]bool) []treeRow {
	var out []treeRow
	for si, st := range b.Stages {
		out = append(out, treeRow{Kind: rowStage, Stage: si, Name: st.Name, State: st.State, Duration: st.Duration})
		if !expanded[st.Name] {
			continue
		}
		for ji, j := range st.Jobs {
			out = append(out, treeRow{Kind: rowJob, Stage: si, Job: ji, Name: j.Name,
				State: j.State, Duration: j.Duration, Key: j.Key, URL: j.URL})
		}
	}
	return out
}

// defaultExpanded opens the stages that failed or are still running, because
// that is what the user opened the build to read.
func defaultExpanded(b provider.Build) map[string]bool {
	out := map[string]bool{}
	for _, st := range b.Stages {
		if st.State == provider.StateFailed || st.State == provider.StateRunning {
			out[st.Name] = true
		}
	}
	return out
}

// detailBody is the main panel's contents for a build: a header, the tree,
// and the failed-test count when the server reported one.
func (m Model) detailBody(width, height int) string {
	b := *m.detail
	var lines []string
	lines = append(lines,
		fmt.Sprintf("%s   #%d   branch %s   %s", stateCell(b.State), b.Number, branchOrDefault(b), view.Duration(b.Duration)),
		dimStyle.Render(strings.TrimSpace(b.Reason+"  "+revisionShort(b))),
		"")

	rows := buildTree(b, m.expanded)
	for i, r := range rows {
		indent := ""
		if r.Kind == rowJob {
			indent = "  "
		}
		cursor := m.cursorFor(m.focus == focusMain, m.treeCursor == i)
		dur := ""
		if r.Duration > 0 {
			dur = view.Duration(r.Duration)
		}
		line := fmt.Sprintf("%s%s%s %-*s %s", cursor, indent,
			stateStyle(r.State).Render(stateGlyph(r.State)),
			maxNameWidth(width), truncate(r.Name, maxNameWidth(width)), dur)
		lines = append(lines, strings.TrimRight(line, " "))
	}
	if n := len(b.FailedTests); n > 0 {
		lines = append(lines, "", errorStyle.Render(fmt.Sprintf("%d failed tests", n)))
	}
	if len(lines) > height {
		lines = lines[:height]
	}
	return strings.Join(lines, "\n")
}

func maxNameWidth(panelWidth int) int {
	w := panelWidth - 22
	if w < 8 {
		w = 8
	}
	return w
}

func revisionShort(b provider.Build) string {
	if len(b.Revisions) == 0 {
		return ""
	}
	return "· " + b.Revisions[0].Short()
}
```

```go
// moveFocused moves exactly one cursor: the focused panel's.
func (m *Model) moveFocused(delta int) {
	switch m.focus {
	case focusPlans:
		m.plans.move(delta)
	case focusBuilds:
		m.builds.move(delta)
	case focusPresets:
		m.presets.move(delta)
	case focusMain:
		m.treeCursor += delta
		m.clampTree()
	}
}

func (m *Model) jumpFocused(top bool) {
	switch m.focus {
	case focusPlans:
		if top { m.plans.top() } else { m.plans.bottom() }
	case focusBuilds:
		if top { m.builds.top() } else { m.builds.bottom() }
	case focusPresets:
		if top { m.presets.top() } else { m.presets.bottom() }
	case focusMain:
		if top {
			m.treeCursor = 0
		} else {
			m.treeCursor = len(m.treeRows()) - 1
		}
		m.clampTree()
	}
}

func (m *Model) clampTree() {
	n := len(m.treeRows())
	if m.treeCursor >= n {
		m.treeCursor = n - 1
	}
	if m.treeCursor < 0 {
		m.treeCursor = 0
	}
}

func (m Model) treeRows() []treeRow {
	if m.detail == nil {
		return nil
	}
	return buildTree(*m.detail, m.expanded)
}

func (m Model) selectedTreeRow() (treeRow, bool) {
	rows := m.treeRows()
	if m.treeCursor < 0 || m.treeCursor >= len(rows) {
		return treeRow{}, false
	}
	return rows[m.treeCursor], true
}
```

```go
// before. At most one watch runs, so the UI makes at most one build's worth

func (m Model) startWatch(key string) (tea.Model, tea.Cmd) {
```

```go
func (m Model) handleOverlayKey(msg tea.KeyMsg) (tea.Model, tea.Cmd) {
	switch {
	case key.Matches(msg, keys.Back):
		return m.back()
	case key.Matches(msg, keys.Quit):
		return m.quit()
	case key.Matches(msg, keys.Down):
		m.picker.move(1)
	case key.Matches(msg, keys.Up):
		m.picker.move(-1)
	case key.Matches(msg, keys.Top):
		m.picker.top()
	case key.Matches(msg, keys.Bottom):
		m.picker.bottom()
	case key.Matches(msg, keys.Enter):
		return m.chooseOverlay()
	}
	return m, nil
}

func (m Model) chooseOverlay() (tea.Model, tea.Cmd) {
	it, ok := m.picker.selected()
	if !ok {
		m.overlay = overlayNone
		return m, nil
	}
	switch m.overlay {
	case overlayServers:
		m.overlay = overlayNone
		if it.Value == m.server {
			return m, nil
		}
		return m.switchServer(it.Value)
	}
	m.overlay = overlayNone
	return m, nil
}
```

```go
type buildLoadedMsg struct{ Build provider.Build }

// reloadBuildCmd re-reads one build. It goes through Service.Build rather
// than the provider directly, so a build the server has dropped is forgotten
// the same way the commands forget it.
func reloadBuildCmd(ctx context.Context, svc *app.Service, key string) tea.Cmd {
	return func() tea.Msg {
		b, err := svc.Build(ctx, key)
		if err != nil {
			return errMsg{Err: err, Where: "build"}
		}
		return buildLoadedMsg{Build: b}
	}
}
```

```go
// refresh reloads whichever panel has focus. It is also the documented retry
// after a watch error, so it clears the error first.
func (m Model) refresh() (tea.Model, tea.Cmd) {
	m.err = nil
	switch m.focus {
	case focusPresets:
		if m.deps.Targets == nil {
			return m, nil
		}
		return m, loadPresetsCmd(m.deps)
	}
	if m.svc == nil {
		return m, nil
	}
	switch m.focus {
	case focusPlans:
		m.plans.loading = true
		return m, loadPlansCmd(context.Background(), m.svc, m.project)
	case focusBuilds:
		if m.buildsPlan == "" {
			return m, nil
		}
		m.builds.loading = true
		return m, loadBuildsCmd(context.Background(), m.svc, m.buildsPlan, buildsPerPlan)
	case focusMain:
		if m.detail == nil {
			return m, nil
		}
		return m, reloadBuildCmd(context.Background(), m.svc, m.detail.Key)
	}
	return m, nil
}
```

and the message case, which keeps the tree as the user left it:

```go
	case buildLoadedMsg:
		b := msg.Build
		m.detail = &b
		m.clampTree()
		return m, nil
```

Note `TestRefreshWithoutAConnectionDoesNothing` uses the default focus, `focusPlans`, so the `m.svc == nil` guard must come before the plans branch — which is why the Presets branch, the only one that does not need a connection, is handled first.
